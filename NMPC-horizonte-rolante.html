k_total = 15     # Número total de passos na simulação real
k_pred = 5       # Horizonte de predição (janela do MPC)
nx = 2           # Dimensão do estado (X, Y) 
nu = 2           # Dimensão do controle (U_x, U_y)
nlambda = 4      # Número de multiplicadores de Lagrange

dt = 1           # Tamanho do passo de tempo 

# Parâmetros geométricos das restrições KKT 
A_kkt = np.array([[ 1,  0], [-1,  0], [ 0,  1], [ 0, -1]])
b_kkt = np.array([1.0, 1.0, 1.0, 1.0])

# Coordenadas globais
x0 = np.array([-5.0, 0.0])
xG = np.array([5.0, 0.0])

Laço do horizonte rolante com criação do modelo e restrições

for t in range(k_total):
    x_atual = x_real[t, :]
    
    # Cria o modelo para a iteração atual
    m = gp.Model(f"NMPC_Step_{t}")
    m.setParam('OutputFlag', 0)     # Desativa o log verboso do Gurobi
    m.Params.NonConvex = 2          # Habilita o solver Não-Convexo para Bilinearidades
    
    # Variáveis de decisão para o horizonte de predição atual
    x_pred = m.addMVar((k_pred + 1, nx), lb=-6, ub=6, name="x_pred")
    xb_pred = m.addMVar((k_pred + 1, nx), lb=-20, ub=20, name="xb_pred")
    u_pred = m.addMVar((k_pred, nu), lb=-1, ub=1, name="u_pred")
    
    lam = m.addMVar((k_pred, nlambda), lb=0, name="lambda")
    mu = m.addMVar((k_pred, nx), lb=-GRB.INFINITY, name="mu")
    
    # FUNÇÃO OBJETIVO (para o horizonte atual)

    # No cálculo da função objetivo (dentro do laço), atribuímos pesos para os termos de erro e controle, nesse

    w1 = 5
    w2 = 10
    obj = sum(xb_pred[k] @ xb_pred[k] for k in range(1, k_pred + 1)) + \
        sum(w1 * (u_pred[k] @ u_pred[k]) for k in range(k_pred))

    m.setObjective(obj, GRB.MINIMIZE)
    
    # RESTRIÇÕES
    # Condição inicial baseada no estado real medido no instante t
    m.addConstr(x_pred[0] == x_atual)
    
    for k in range(k_pred):
        # Cálculo do erro em relação ao alvo global xG
        m.addConstr(xb_pred[k] == x_pred[k] - xG)
        
        # Dinâmica de transição de estados
        m.addConstr(x_pred[k+1] == x_pred[k] + u_pred[k] * dt)
        
        # Restrições Geométricas / KKT Não-Convexas do seu modelo base
        m.addConstr(-lam[k] @ b_kkt + mu[k] @ x_pred[k] >= 1)
        m.addConstr(A_kkt.T @ lam[k] - mu[k] == np.zeros(nx))
        m.addConstr((mu[k] @ mu[k]) <= 1)
        
    # Último cálculo de erro para o ponto final do horizonte
    m.addConstr(xb_pred[k_pred] == x_pred[k_pred] - xG)
    
    # Executa a otimização do passo
    m.optimize()
    
    # Processa os resultados do passo atual
    if m.status != GRB.OPTIMAL:
        print(f"Otimização falhou no passo t={t}. Status: {m.status}. Aplicando controle nulo.")
        u_aplicar = np.zeros(nu)
        previsoes_salvas.append(None)
    else:
        # Extrai o primeiro controle ótimo calculado para aplicar ao sistema real
        u_plano = u_pred.X
        u_aplicar = u_plano[0, :]
        previsoes_salvas.append(x_pred.X)
        
    # Aplica o controle no sistema real simulado
    u_real[t, :] = u_aplicar
    x_real[t+1, :] = x_real[t, :] + u_aplicar * dt
    
    print(f"Passo t={t:02d} | Estado Atual: [{x_atual[0]:.2f}, {x_atual[1]:.2f}] | Controle Aplicado: [{u_aplicar[0]:.2f}, {u_aplicar[1]:.2f}]")

print("\nSimulação NMPC finalizada!")
